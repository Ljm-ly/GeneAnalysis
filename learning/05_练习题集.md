```python
# 写一个 Calculator 类
# __init__ 接收 initial_value（默认0）
# 方法：
#   add(n) —— 加上n
#   subtract(n) —— 减去n
#   multiply(n) —— 乘以n
#   divide(n) —— 除以n（除数为0时打印警告，返回None）
#   get_result() —— 返回当前结果

# 你的代码
```

<details>
<summary>参考答案</summary>

```python
class Calculator:
    def __init__(self, initial_value=0):
        self.result = initial_value
    
    def add(self, n):
        self.result += n
        return self.result
    
    def subtract(self, n):
        self.result -= n
        return self.result
    
    def multiply(self, n):
        self.result *= n
        return self.result
    
    def divide(self, n):
        if n == 0:
            print("警告：不能除以零！")
            return None
        self.result /= n
        return self.result
    
    def get_result(self):
        return self.result

# 测试

calc.subtract(3)   # 12
calc.multiply(2)   # 24
calc.divide(4)     # 6.0
```

### 练习5-3：图书类

```python
# 写一个 Book 类
# __init__ 接收 title, author, price
# 属性：is_borrowed（初始False）
# 方法：
#   borrow() —— 借书（如果已借出打印"已被借走"）
#   return_book() —— 还书
#   get_info() —— 返回图书信息字符串

# 你的代码
```

<details>
<summary>参考答案</summary>

```python
class Book:
    def __init__(self, title, author, price):
        self.title = title
        self.author = author
        self.price = price
        self.is_borrowed = False
    
    def borrow(self):
        if self.is_borrowed:
            print(f"《{self.title}》已被借走")
        else:
            self.is_borrowed = True
            print(f"《{self.title}》借出成功")
    
    def return_book(self):
        self.is_borrowed = False
        print(f"《{self.title}》已归还")
    
    def get_info(self):
        status = "已借出" if self.is_borrowed else "在馆"
        return f"《{self.title}》作者：{self.author}，{self.price}元，{status}"

# 测试
book = Book("Python入门", "张三", 59)
print(book.get_info())
book.borrow()
print(book.get_info())
book.return_book()
```
</details>

```python
    try:
        with open(filename, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        print("文件不存在")
        return ""

content = safe_read("test.txt")
```

```python
print(cleaner.remove_duplicates(data))  # [3, 1, 2, 5]

print(cleaner.normalize([10, 20, 30]))  # [0.0, 0.5, 1.0]
```

<details>
<summary>参考答案</summary>

```python
class ListCleaner:
    def remove_duplicates(self, data):
        seen = []
        for item in data:
            if item not in seen:
                seen.append(item)
        return seen
    
    def remove_none(self, data):
        return [x for x in data if x is not None]
    
    def normalize(self, data):
        min_val = min(data)
        max_val = max(data)
        if max_val == min_val:
            return [0.0] * len(data)
        return [(x - min_val) / (max_val - min_val) for x in data]
```
</details>
