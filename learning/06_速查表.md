## 数据类型速查

```python
# 整数
n = 100

# 浮点数
f = 3.14

# 字符串
s = "hello"
s.upper()       # "HELLO"
s.lower()       # "hello"
s.split(",")    # ["a", "b"]
len(s)          # 5
f"value={n}"    # "value=100"

# 布尔
flag = True  # 或 False

# 列表
lst = [1, 2, 3]
lst[0]          # 1
lst[-1]         # 3
lst.append(4)   # [1, 2, 3, 4]
lst[0] = 10     # [10, 2, 3, 4]
len(lst)        # 4

# 字典
d = {"a": 1, "b": 2}
d["a"]          # 1
d["c"] = 3      # 添加
d.keys()        # ["a", "b", "c"]
d.values()      # [1, 2, 3]

# 元组（不可变）
t = (10, 20)
t[0]            # 10
# t[0] = 30     # 报错！
```

## 条件判断速查

```python
if 条件1:
    ...
elif 条件2:
    ...
else:
    ...

# 逻辑运算符
a and b    # 两个都为真
a or b     # 至少一个为真
not a      # 取反

# 比较运算符
==   # 等于
!=   # 不等于
>    # 大于
<    # 小于
>=   # 大于等于
<=   # 小于等于
```

## 循环速查

```python
# for 循环
for item in 列表:
    ...

for i in range(5):      # 0,1,2,3,4
    ...

for i in range(1, 6):   # 1,2,3,4,5
    ...

for i, item in enumerate(列表):
    # i 是索引，item 是元素
    ...

# while 循环
while 条件:
    ...

# 控制
break      # 跳出循环
continue   # 跳过当前次
```

## 函数速查

```python
# 定义
def 函数名(参数1, 参数2=默认值):
    """文档字符串"""
    ...
    return 结果

# 调用
结果 = 函数名(值1, 参数2=值2)

# 列表推导式
[x*2 for x in 列表]
[x for x in 列表 if x > 0]

# zip
for a, b in zip(列表1, 列表2):
    ...
```

## 类速查

```python

    def __init__(self, 参数):

    def 方法名(self, 参数):
        """方法说明"""
        ...

# 访问属性
对象.属性

# 调用方法
对象.方法名(参数)
```
